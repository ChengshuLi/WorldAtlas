import json, re
from pathlib import Path
from datetime import datetime
D=Path(__file__).parent
def duration(start,end):
 a,b=(datetime.fromisoformat(s.replace('Z','+00:00')) for s in (start,end))
 if a.tzinfo is None or b.tzinfo is None or b<a: raise ValueError('Expected timezone-aware ordered timestamps')
 return (b-a).total_seconds()
def write(name,value): (D/name).write_text(json.dumps(value,indent=2)+'\n')
cases=[]
for n in [0,9,147,171]:
 end='2026-10-04T00:%02d:%02dZ'%(n//60,n%60)
 actual=duration('2026-10-04T00:00:00Z',end);assert actual==n
 cases.append({'expected_seconds':n,'actual_seconds':actual})
write('positive-control.json',{'method_id':'hosted-budget-observation','kind':'positive-control','outcome':'passed','cases':cases})
try:duration('2026-10-04T00:00:09Z','2026-10-04T00:00:00Z')
except ValueError as e:write('negative-control.json',{'method_id':'hosted-budget-observation','kind':'negative-control','outcome':'passed','inverted_timestamps_rejected':True,'error':str(e)})
else:raise AssertionError('Invalid duration accepted')
samples={}
for name in ['before-package','full-package','fast-package','fast-classify']:

 samples[name]={'job_id':j['id'],'head_sha':j['head_sha'],'started_at':j['started_at'],'completed_at':j['completed_at'],'seconds':duration(j['started_at'],j['completed_at'])}

  found=re.findall(r'ℹ '+key+r' (\d+)\s*$',text,re.M);assert len(found)==1,(name,key,found);counts[key]=int(found[0])

 samples[name]['full_build_preserved' if full else 'install_and_build_skipped']=True

assert len(decisions)==1 and decisions[0]['full'] is False
